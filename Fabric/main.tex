\begin{lstlisting}[language=Python]
# DR drill: restore an overwritten/deleted Gold table via Delta time travel
before = spark.read.table("gold.sales_summary").count()   # baseline

spark.sql("DELETE FROM gold.sales_summary WHERE sale_date < '2024-01-01'")
print("after delete:", spark.read.table("gold.sales_summary").count())

# Option 1: restore the table to a previous version
spark.sql("RESTORE TABLE gold.sales_summary TO VERSION AS OF 3")

# Option 2: inspect a point-in-time snapshot without restoring
# spark.sql("SELECT COUNT(*) FROM gold.sales_summary "
#           "TIMESTAMP AS OF '2024-06-01T00:00:00Z'").show()

restored = spark.read.table("gold.sales_summary").count()
assert restored == before, f"DRILL FAIL: {restored} != baseline {before}"
print("restore verified:", restored)
\end{lstlisting}
    \dayitem{Friday}{\usecase{Troubleshoot and re-architect a severely degraded Delta table that has accumulated millions of small files (small file problem) over 6 months of streaming.}}
\end{itemize}
\prodnote{Delta time travel is your first line of accidental-delete recovery in Fabric: OneLake has no traditional backup/restore, so a bad overwrite or dropped partition is recovered with RESTORE or VERSION AS OF against Delta history. Pair it with soft-delete patterns (tombstone flags) for row-level mistakes, and watch your VACUUM retention --- vacuuming below the retention horizon destroys the very history you rely on.}

\subsection*{Milestone 1 Capstone: Reference Solution Sketch}
\begin{itemize}[label=-, itemsep=0pt]
    \item \textbf{Architecture:} OneLake shortcuts $\rightarrow$ Bronze Lakehouse (raw files) $\rightarrow$ parameterized Spark notebooks $\rightarrow$ Silver (validated Delta) $\rightarrow$ Gold star schema; chosen for zero-copy multi-cloud ingestion and engine-agnostic Delta storage.
    \item Create three Lakehouses (Bronze/Silver/Gold) and shortcut external data into Bronze --- no copy jobs.
    \item Parameterize notebooks by source path and batch date; run Bronze $\rightarrow$ Silver $\rightarrow$ Gold in order from a pipeline.
    \item Write Silver with dedup on the business key, explicit schemas, and Delta schema enforcement.
    \item Build Gold as fact + conformed dimensions; enable V-Order and OPTIMIZE with Z-ORDER on join keys.
    \item Capture Monitoring Hub runs and Delta history for the time-travel recovery demo.
\end{itemize}
\begin{lstlisting}[language=Python]
# Gold optimization: compact files and co-locate join keys
spark.sql("OPTIMIZE gold.fact_sales ZORDER BY (customer_key)")
spark.sql("OPTIMIZE gold.dim_customer ZORDER BY (customer_key)")
# Verify compaction took effect
(spark.sql("DESCRIBE DETAIL gold.fact_sales")
      .select("numFiles", "sizeInBytes").show())
\end{lstlisting}
\begin{itemize}[label=-, itemsep=0pt]
    \item \textbf{Pitfall:} \texttt{inferSchema} on large CSVs scans everything and mis-infers --- declare explicit schemas.
    \item \textbf{Pitfall:} small-file buildup from frequent appends; schedule OPTIMIZE, do not run it ad hoc.
    \item \textbf{Pitfall:} \texttt{spark.read.table} only sees the attached Lakehouse by default --- use the SQL endpoint or shortcuts for cross-Lakehouse reads.
\end{itemize}

\begin{lstlisting}[language=Python]
# SCD Type 2 upsert into gold.dim_customer via Delta MERGE (PySpark)
from delta.tables import DeltaTable

tgt = DeltaTable.forName(spark, "gold.dim_customer")
src = spark.read.table("silver.customer_daily")  # today's snapshot

# Step 1: expire current rows whose tracked attributes changed
(tgt.alias("t").merge(src.alias("s"),
    "t.customer_id = s.customer_id AND t.is_current = true")
 .whenMatchedUpdate(
    condition="t.segment <> s.segment OR t.region <> s.region",
    set={"is_current": "false", "valid_to": "s.change_date"})
 .execute())

# Step 2: insert new versions (changed attributes + brand-new customers)
(tgt.alias("t").merge(src.alias("s"),
    "t.customer_id = s.customer_id AND t.is_current = true"
    " AND t.segment = s.segment AND t.region = s.region")
 .whenNotMatchedInsert(values={
    "customer_id": "s.customer_id", "segment": "s.segment",
    "region": "s.region", "is_current": "true",
    "valid_from": "s.change_date", "valid_to": "null"})
 .execute())
\end{lstlisting}

# Data-quality assertions before writing to Gold
from pyspark.sql import functions as F

df = spark.read.table("silver.orders")

null_pk  = df.filter(F.col("order_id").isNull()).count()
dup_keys = (df.groupBy("order_id").count()
              .filter("count > 1").count())

assert null_pk == 0,  f"DQ FAIL: {null_pk} null order_id values"
assert dup_keys == 0, f"DQ FAIL: {dup_keys} duplicate order_id keys"

# Quarantine bad rows instead of failing silently
rejected = df.filter(F.col("amount") < 0)
if rejected.count() > 0:
    (rejected.withColumn("reject_reason", F.lit("negative_amount"))
             .write.format("delta").mode("append")
             .saveAsTable("gold.rejected_orders"))

(df.filter(F.col("amount") >= 0)
   .write.format("delta").mode("overwrite")
   .saveAsTable("gold.orders"))
\end{lstlisting}

\begin{lstlisting}[language=SQL]
// 5-minute rolling average + anomaly detection per device
IoTTelemetry
| where Timestamp > ago(6h)
| make-series AvgTemp = avg(Temperature) default=0
              on Timestamp step 5m by DeviceId
| extend RollingAvg = series_fir(AvgTemp, repeat(1, 5))
| extend Anomalies = series_decompose_anomalies(RollingAvg, 1.5)
| mv-expand Timestamp, RollingAvg, Anomalies
| project DeviceId,
          Timestamp = todatetime(Timestamp),
          RollingAvg = todouble(RollingAvg),
          IsAnomaly = toint(Anomalies) == 1
| order by DeviceId, Timestamp
\end{lstlisting}
    \dayitem{Friday}{End-to-end exactly-once: Eventstream at-least-once delivery, idempotent ingestion into KQL/Lakehouse, dedup-key placement, and downstream consumer checkpointing. \usecase{Troubleshoot a scenario where a streaming dashboard is showing inaccurate daily aggregations due to mobile devices going offline and uploading data late.}}
\begin{lstlisting}[language=SQL]
// Dedup (keep latest arrival per id), windowed stream join, dimension lookup
let Telemetry =
    IoTTelemetry
    | where Timestamp > ago(1h)              // bounded lookback window
    | summarize arg_max(ingestion_time(), *) by EventId;
let Alerts =
    DeviceAlerts
    | where AlertTime > ago(1h)
    | summarize arg_max(ingestion_time(), *) by AlertId;
Telemetry
| join kind=inner Alerts on DeviceId
| where abs(datetime_diff('second', Timestamp, AlertTime)) <= 300
| join kind=inner (                          // stream-to-dimension
    Devices | project DeviceId, Site, Model
) on DeviceId
| summarize Readings = count() by Site, Model, bin(Timestamp, 1m)
\end{lstlisting}
\end{itemize}
\prodnote{Eventstream replay is the top source of silent duplicates downstream: after a consumer failure or reconfiguration, Eventstream redelivers events from the source's retention window, and every at-least-once hop (Eventstream $\rightarrow$ KQL, Eventstream $\rightarrow$ Lakehouse) can double-land rows. Stamp the dedup key at the producer (a unique \texttt{event\_id} generated before the first send, not at any hop), propagate it untouched through Eventstream, and enforce dedup at the first queryable boundary --- a KQL materialized view with \texttt{arg\_max}, or a Delta MERGE on \texttt{event\_id} for the Lakehouse --- so every downstream consumer (dashboards, Activator rules, reverse ETL) reads already-clean data. Pair this with consumer checkpointing: record the last processed event time/offset so restarts resume instead of replaying.}

\section{Appendix A: Glossary}
\begin{description}[leftmargin=3.2cm, style=nextline]
    \item[OneLake] The single, tenant-wide data lake underlying all of Fabric (``OneDrive for data'').
    \item[Lakehouse] A Fabric item combining files and managed Delta tables with a SQL analytics endpoint.
    \item[Data Warehouse] Fabric's T-SQL-native analytical store with full transactional DDL/DML support.
    \item[Delta Lake] Open table format (Parquet + transaction log) used by all Fabric engines.
    \item[Medallion Architecture] Bronze (raw) $\rightarrow$ Silver (validated) $\rightarrow$ Gold (enriched) layering pattern.
    \item[Shortcut] A OneLake pointer that virtualizes external data (ADLS, S3, Dataverse) without copying it.
    \item[Semantic Model] The Power BI dataset layer defining tables, relationships, and DAX measures.
    \item[DirectLake] Power BI storage mode reading Delta/Parquet directly into memory without import or DirectQuery.
    \item[SQL Analytics Endpoint] The automatic read-only T-SQL endpoint generated for every Lakehouse.
    \item[Dataflow Gen2] Power Query Online-based visual ETL with output destinations.
    \item[Pipeline] Fabric Data Factory orchestration item (Copy, ForEach, Notebook activities, etc.).
    \item[Eventstream] Real-time ingestion and routing item for streaming events.
    \item[Eventhouse] Fabric's real-time analytics store hosting one or more KQL databases.
    \item[KQL] Kusto Query Language for time-series and log analytics over Eventhouses.
    \item[Data Activator (Reflex)] No-code item that triggers actions when data meets conditions.
    \item[MLflow] Open-source experiment tracking and model registry framework integrated into Fabric.
    \item[SynapseML] Microsoft's distributed ML library for Spark, including Azure AI service connectors.
    \item[Capacity Unit (CU)] Fabric's compute billing unit; capacities are sized in SKUs (F2--F2048).
    \item[Throttling] Capacity protection that delays or rejects operations when CU consumption exceeds limits.
    \item[Workspace Role] Admin, Member, Contributor, or Viewer --- the coarse access boundary in Fabric.
    \item[Git Integration] Native sync of workspace items to Azure DevOps/GitHub repositories.
    \item[Deployment Pipeline] Fabric item promoting content across Dev $\rightarrow$ Test $\rightarrow$ Prod workspaces.
    \item[Sensitivity Label] Microsoft Purview classification applied to Fabric items, inherited downstream.
    \item[Monitoring Hub] Central Fabric surface for run history, status, and alert routing.
    \item[Purview Hub] In-Fabric governance portal for lineage, endorsement, and tenant insights.
\end{description}

\noindent\textbf{Week 14:}
\begin{enumerate}[leftmargin=8mm, itemsep=0pt]
    \item It stores only the aggregated delta per new ingested batch instead of rescanning raw rows; \texttt{arg\_max} collapses replayed copies of the same \texttt{EventId}, keeping the latest arrival by \texttt{ingestion\_time()}.
    \item State and scan cost grow without bound: every query/materialization cycle re-reads more history, raising memory, CPU, and latency until the Eventhouse falls behind the ingest rate.
    \item Join on the shared key, then filter on \texttt{datetime\_diff} between the two event times within the window; late arrivals fall outside the window and are dropped or mis-matched, skewing the joined aggregates.
    \item Generated at the producer before the first send and enforced at the first queryable boundary (KQL materialized view or Delta MERGE on \texttt{event\_id}), so replay never duplicates rows downstream.
    \item It records the last processed event time/offset so a restarted consumer resumes where it stopped; without it, restarts replay events and re-emit duplicates.
\end{enumerate}
